How this page is built 
// Server: holds the API key
const vny = new Vnytros({
  apiKey: process.env.VNYTROS_API_KEY,
  baseUrl: process.env.VNYTROS_BASE_URL, // your own deployment
});
app.get("/api/detect", async (req, res) => {
  res.json(await vny.detect(req.query.domain));
});
app.post("/api/status", async (req, res) => {
  // { domain, ip, target, txt } -> is_configured + per-record status
  res.json(await vny.domains.status(req.body));
});
